def read_csv(path):
    return (
        spark.read
        .option("header", "true")
        .option("inferSchema", "true")
        .option("mode", "PERMISSIVE")
        .csv(path)
    )


def write_delta(df, path):
    (
        df.write
        .format("delta")
        .mode("overwrite")
        .option("overwriteSchema", "true")
        .save(path)
    )


def clean_string(column_name):
    return F.trim(F.col(column_name))


def clean_upper(column_name):
    return F.upper(F.trim(F.col(column_name)))


def clean_lower(column_name):
    return F.lower(F.trim(F.col(column_name)))

df_orders_raw = (
    spark.read
    .format("csv")
    .option("header", "true")
    .option("inferSchema", "false")
    .option("mode", "PERMISSIVE")
    .option("quote", '"')
    .option("escape", '"')
    .option("multiLine", "true")
    .option("ignoreLeadingWhiteSpace", "true")
    .option("ignoreTrailingWhiteSpace", "true")
    .csv(f"{bronze_path}/olist_orders_dataset")
)

df_orders = (
    df_orders_raw
    .select(
        F.trim(F.col("order_id")).alias("order_id"),
        F.trim(F.col("customer_id")).alias("customer_id"),

        F.lower(
            F.trim(F.col("order_status"))
        ).alias("order_status"),

        F.expr("""
            try_to_timestamp(
                trim(order_purchase_timestamp),
                'yyyy-MM-dd HH:mm:ss'
            )
        """).alias("order_purchase_timestamp"),

        F.expr("""
            try_to_timestamp(
                trim(order_approved_at),
                'yyyy-MM-dd HH:mm:ss'
            )
        """).alias("order_approved_at"),

        F.expr("""
            try_to_timestamp(
                trim(order_delivered_carrier_date),
                'yyyy-MM-dd HH:mm:ss'
            )
        """).alias("order_delivered_carrier_date"),

        F.expr("""
            try_to_timestamp(
                trim(order_delivered_customer_date),
                'yyyy-MM-dd HH:mm:ss'
            )
        """).alias("order_delivered_customer_date"),

        F.expr("""
            try_to_timestamp(
                trim(order_estimated_delivery_date),
                'yyyy-MM-dd HH:mm:ss'
            )
        """).alias("order_estimated_delivery_date")
    )
    .filter(
        F.col("order_id").isNotNull() &
        F.col("customer_id").isNotNull()
    )
    .dropDuplicates(["order_id"])
)

write_delta(
    df_orders,
    f"{silver_path}/orders"
)

# PAYMENTS
# GRÃO: 1 LINHA = 1 REGISTRO DE PAGAMENTO DO PEDIDO
# ============================================================

df_payments_raw = (
    spark.read
    .format("csv")
    .option("header", "true")
    .option("inferSchema", "false")
    .option("mode", "PERMISSIVE")
    .option("quote", '"')
    .option("escape", '"')
    .option("multiLine", "true")
    .option("ignoreLeadingWhiteSpace", "true")
    .option("ignoreTrailingWhiteSpace", "true")
    .csv(f"{bronze_path}/olist_order_payments_dataset")
)

df_payments = (
    df_payments_raw
    .select(
        F.trim(F.col("order_id")).alias("order_id"),

        F.expr(
            "try_cast(trim(payment_sequential) as int)"
        ).alias("payment_sequential"),

        F.lower(
            F.trim(F.col("payment_type"))
        ).alias("payment_type"),

        F.expr(
            "try_cast(trim(payment_installments) as int)"
        ).alias("payment_installments"),

        F.expr(
            "try_cast(trim(payment_value) as double)"
        ).alias("payment_value")
    )

    # Chaves mínimas obrigatórias
    .filter(
        F.col("order_id").isNotNull() &
        F.col("payment_sequential").isNotNull()
    )

    # Uma linha por pagamento/sequência
    .dropDuplicates([
        "order_id",
        "payment_sequential"
    ])

    # Indicador de qualidade
    .withColumn(
        "payment_installments_valido",
        F.when(
            F.col("payment_installments").isNull(),
            F.lit(False)
        )
        .when(
            F.col("payment_installments") >= 1,
            F.lit(True)
        )
        .otherwise(
            F.lit(False)
        )
    )
)

write_delta(
    df_payments,
    f"{silver_path}/payments"
)

print(f"✅ payments: {df_payments.count():,} registros")

df_customers_raw = (
    spark.read
    .format("csv")
    .option("header", "true")
    .option("inferSchema", "false")
    .option("mode", "PERMISSIVE")
    .option("quote", '"')
    .option("escape", '"')
    .option("multiLine", "true")
    .option("ignoreLeadingWhiteSpace", "true")
    .option("ignoreTrailingWhiteSpace", "true")
    .csv(f"{bronze_path}/olist_customers_dataset")
)

df_customers = (
    df_customers_raw
    .select(
        F.trim(
            F.col("customer_id")
        ).alias("customer_id"),

        F.trim(
            F.col("customer_unique_id")
        ).alias("customer_unique_id"),

        # Mantemos o CEP com 5 dígitos
        F.lpad(
            F.regexp_replace(
                F.trim(
                    F.col("customer_zip_code_prefix")
                ),
                "[^0-9]",
                ""
            ),
            5,
            "0"
        ).alias("customer_zip_code_prefix"),

        # Padronização textual
        F.upper(
            F.trim(
                F.col("customer_city")
            )
        ).alias("customer_city"),

        F.upper(
            F.trim(
                F.col("customer_state")
            )
        ).alias("customer_state")
    )

    # Chaves obrigatórias
    .filter(
        F.col("customer_id").isNotNull() &
        F.col("customer_unique_id").isNotNull()
    )

    # 1 registro por customer_id
    .dropDuplicates(["customer_id"])
)
